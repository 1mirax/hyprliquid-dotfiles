# AUR packages at the bottom are the Bluetooth and audio menus; the setup
# works without them, see the note there.

# --- compositor and session ---------------------------------------------
hyprland
hyprlock
hypridle
hyprpaper
hyprcursor
uwsm                        # session manager: runs the daemons as systemd units
# Only the Hyprland backend, on purpose. It implements the compositor-specific
# interfaces - screen capture, global shortcuts - and nothing else. The usual
# companion, xdg-desktop-portal-gtk, would add AppChooser, Print, Notification
# and friends, all of which exist for sandboxed apps that are not installed
# here, plus Settings.
#
# Settings is the one real loss: it is how an application asks the session
# whether to prefer a dark theme over D-Bus. GTK apps do not need it, they read
# gtk-application-prefer-dark-theme from .config/gtk-3.0/settings.ini directly.
# Firefox does use it, so with no backend for it the browser must be told to
# use a dark theme in its own settings. That is the trade: one manual setting
# against a 6.5 MB resident process.
xdg-desktop-portal-hyprland

# --- sound --------------------------------------------------------------
# All four are required. pipewire alone is only the engine: without
# wireplumber nothing claims the hardware and the graph holds two dummy nodes,
# and without pipewire-pulse there is no socket for Firefox and everything
# else that speaks only the PulseAudio protocol.
pipewire
wireplumber                 # session manager - decides what connects to what
pipewire-pulse              # PulseAudio protocol, for Firefox, Electron, Wine
pipewire-alsa               # routes ALSA-only applications into the graph
alsa-utils                  # aplay / amixer, for diagnosing the bottom layer

# --- bluetooth ----------------------------------------------------------
bluez
bluez-utils                 # bluetoothctl

# --- hardware controls --------------------------------------------------
brightnessctl
polkit-gnome                # the authentication agent

# --- shell -------------------------------------------------------------
fish
starship

# --- power management (see .config/power/install.sh) --------------------
tlp
throttled                   # extra/ - undervolt and HWP control for this ThinkPad

gtk-layer-shell             # waybar needs it
